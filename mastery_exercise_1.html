Exercise 2.3: Lab Result Validator#

Write a function that validates lab results and handles errors gracefully.

The function should:

  1. Accept a lab value as a string

  2. Try to convert it to a float

  3. Check if it’s within the normal range (10-20)

  4. Return a tuple: (value, status) where status is "normal", "abnormal", or "invalid"

Test with: ["15.5", "25.3", "N/A", "8.2", "18.0"]

# TODO: Write your code here
# Hint: Use try...except, comparison operators, and return a tuple
🔍 Click to reveal solution

Solution

def validate_lab_result(value_str):
    """Validates a lab result and returns (value, status)"""
    try:
        value = float(value_str)
        if 10 <= value <= 20:
            status = "normal"
        else:
            status = "abnormal"
        return (value, status)
    except ValueError:
        return (None, "invalid")

# Test the function
test_values = ["15.5", "25.3", "N/A", "8.2", "18.0"]

print("Lab Result Validation:")
for test in test_values:
    value, status = validate_lab_result(test)
    if value is not None:
        print(f"  {test} -> {value} ({status})")
    else:
        print(f"  {test} -> {status}")

Expected output:

Lab Result Validation:
  15.5 -> 15.5 (normal)
  25.3 -> 25.3 (abnormal)
  N/A -> invalid
  8.2 -> 8.2 (abnormal)
  18.0 -> 18.0 (normal)

Exercise 2.5: Blood Pressure Readings Analysis#

Given a list of blood pressure tuples (systolic, diastolic): [(120, 80), (145, 92), (118, 75), (160, 95), (125, 82)]

Calculate and print:

  1. Average systolic pressure

  2. Average diastolic pressure

  3. Number of readings with high systolic (≥140)

  4. The reading with the highest systolic pressure

# TODO: Write your code here
# Hint: Use loops, tuple unpacking, and the max() function
🔍 Click to reveal solution

Solution

bp_readings = [(120, 80), (145, 92), (118, 75), (160, 95), (125, 82)]

# Calculate averages
systolic_sum = 0
diastolic_sum = 0
high_count = 0

for systolic, diastolic in bp_readings:
    systolic_sum += systolic
    diastolic_sum += diastolic
    if systolic >= 140:
        high_count += 1

avg_systolic = systolic_sum / len(bp_readings)
avg_diastolic = diastolic_sum / len(bp_readings)

# Find highest reading
highest_reading = max(bp_readings, key=lambda x: x[0])

# Print results
print(f"Average Systolic: {avg_systolic:.1f} mmHg")
print(f"Average Diastolic: {avg_diastolic:.1f} mmHg")
print(f"High readings (≥140): {high_count}")
print(f"Highest reading: {highest_reading[0]}/{highest_reading[1]} mmHg")

Exercise 2.10: Safe Division with Multiple Checks#

Write a function that calculates a ratio with comprehensive error handling.

The function should:

  1. Accept two string parameters (numerator and denominator)

  2. Try to convert both to floats

  3. Perform division

  4. Handle ValueError (invalid conversion) and ZeroDivisionError

  5. Return the result or an error message

Test with:

  • ("100", "5") → should work

  • ("100", "0") → zero division

  • ("100", "N/A") → invalid value

# TODO: Write your code here
# Hint: Use nested try...except blocks or multiple except clauses
🔍 Click to reveal solution

Solution

def safe_ratio(num_str, denom_str):
    """Calculate ratio with comprehensive error handling"""
    try:
        numerator = float(num_str)
        denominator = float(denom_str)
        result = numerator / denominator
        return f"Result: {result:.2f}"
    except ValueError:
        return "Error: Invalid number format"
    except ZeroDivisionError:
        return "Error: Cannot divide by zero"

# Test cases
test_cases = [
    ("100", "5"),
    ("100", "0"),
    ("100", "N/A")
]

for num, denom in test_cases:
    print(f"{num} / {denom}: {safe_ratio(num, denom)}")

Expected output:

100 / 5: Result: 20.00
100 / 0: Error: Cannot divide by zero
100 / N/A: Error: Invalid number format

def check_drug_interactions(prescribed_meds, interaction_db):
    """Check for dangerous drug interactions"""
    interactions_found = []
    
    # Check all pairs of medications
    for i in range(len(prescribed_meds)):
        for j in range(i + 1, len(prescribed_meds)):
            drug1 = prescribed_meds[i].lower()
            drug2 = prescribed_meds[j].lower()
            
            # Check both orderings since dictionary keys are ordered tuples
            pair1 = (drug1, drug2)
            pair2 = (drug2, drug1)
            
            if pair1 in interaction_db:
                interactions_found.append((drug1, drug2, interaction_db[pair1]))
            elif pair2 in interaction_db:
                interactions_found.append((drug1, drug2, interaction_db[pair2]))
    
    return interactions_found

def generate_safety_report(prescribed_meds, interactions_found):
    """Generate a safety report for prescribed medications"""
    print(f"\n=== Drug Interaction Report ===")
    print(f"Prescribed medications: {', '.join(prescribed_meds)}")
    print()
    
    if interactions_found:
        print("⚠️  WARNING: Potential interactions detected:")
        for drug1, drug2, risk in interactions_found:
            print(f"  - {drug1.upper()} + {drug2.upper()}: {risk}")
    else:
        print("✓ No known interactions detected")
    print()

Analyze data from a clinical trial with the following requirements:

def analyze_trial_data(patient_data):
    """Analyze clinical trial results"""
    
    improved = 0
    unchanged = 0
    worsened = 0
    dropouts = 0
    total_change = 0
    completers = 0
    
    results = []
    
    for patient in patient_data:
        patient_id = patient["id"]
        baseline = patient["baseline"]
        followup = patient.get("followup")
        
        # Handle dropouts
        if followup is None:
            dropouts += 1
            results.append({
                "id": patient_id,
                "status": "Dropped out",
                "change": None
            })
            continue
        
        # Calculate change (negative means improvement for things like BP, glucose)
        change = followup - baseline
        completers += 1
        total_change += change
        
        # Categorize outcome
        if change <= -10:
            status = "Improved"
            improved += 1
        elif change < 0 or abs(change) < 10:
            status = "Unchanged"
            unchanged += 1
        else:
            status = "Worsened"
            worsened += 1
        
        results.append({
            "id": patient_id,
            "baseline": baseline,
            "followup": followup,
            "change": change,
            "status": status
        })
    
    # Calculate statistics
    total_patients = len(patient_data)
    dropout_rate = (dropouts / total_patients) * 100
    avg_change = total_change / completers if completers > 0 else 0
    
    return {
        "results": results,
        "improved": improved,
        "unchanged": unchanged,
        "worsened": worsened,
        "dropouts": dropouts,
        "dropout_rate": dropout_rate,
        "average_change": avg_change
    }

def analyze_lab_time_series(measurements):
    """Comprehensive analysis of lab values over time"""
    
    valid_readings = []
    missing_days = []
    rapid_changes = []
    
    # Process and validate data
    for day, value in measurements:
        try:
            numeric_value = float(value)
            valid_readings.append((day, numeric_value))
        except (ValueError, TypeError):
            missing_days.append(day)
    
    # Calculate basic statistics
    if len(valid_readings) == 0:
        return "Error: No valid readings found"
    
    values_only = [val for day, val in valid_readings]
    average = sum(values_only) / len(values_only)
    
    # Determine trend
    first_value = values_only[0]
    last_value = values_only[-1]
    change_percent = ((last_value - first_value) / first_value) * 100
    
    if change_percent > 10:
        trend = "Increasing"
    elif change_percent < -10:
        trend = "Decreasing"
    else:
        trend = "Stable"
    
    # Check for rapid changes between consecutive readings
    for i in range(1, len(valid_readings)):
        prev_day, prev_val = valid_readings[i-1]
        curr_day, curr_val = valid_readings[i]
        
        percent_change = abs((curr_val - prev_val) / prev_val) * 100
        
        if percent_change > 15:
            rapid_changes.append({
                "from_day": prev_day,
                "to_day": curr_day,
                "change": curr_val - prev_val,
                "percent": percent_change
            })
    
    return {
        "valid_readings": len(valid_readings),
        "average": average,
        "trend": trend,
        "change_percent": change_percent,
        "missing_days": missing_days,
        "rapid_changes": rapid_changes,
        "readings": valid_readings
    }

def assign_triage_priority(patient, critical_symptoms):
    """Assign triage priority based on symptoms and vitals"""
    
    patient_id = patient.get("id", "Unknown")
    symptoms = patient.get("symptoms", set())
    vitals = patient.get("vitals", {})
    
    # Extract vitals with defaults
    temp = vitals.get("temp", 37.0)
    hr = vitals.get("hr", 70)
    bp_sys = vitals.get("bp_sys", 120)
    
    # Check for critical symptoms
    has_critical = len(symptoms & critical_symptoms) > 0
    
    # Determine priority
    if has_critical or (temp > 39.0 and hr > 120):
        priority = 1
        category = "CRITICAL"
        reason = []
        if has_critical:
            critical_found = symptoms & critical_symptoms
            reason.append(f"Critical symptoms: {', '.join(critical_found)}")
        if temp > 39.0 and hr > 120:
            reason.append(f"High fever + tachycardia")
    
    elif temp > 38.5 or hr > 100 or bp_sys > 160:
        priority = 2
        category = "URGENT"
        reason = []
        if temp > 38.5:
            reason.append(f"Fever: {temp}°C")
        if hr > 100:
            reason.append(f"Elevated HR: {hr} bpm")
        if bp_sys > 160:
            reason.append(f"High BP: {bp_sys} mmHg")
    
    elif len(symptoms) > 0:
        priority = 3
        category = "STANDARD"
        reason = [f"Symptoms present, vitals stable"]
    
    else:
        priority = 4
        category = "NON-URGENT"
        reason = ["Minor complaint, stable vitals"]
    
    return {
        "id": patient_id,
        "priority": priority,
        "category": category,
        "reason": "; ".join(reason),
        "symptoms": symptoms,
        "vitals": vitals
    }

def generate_triage_queue(patients, critical_symptoms):
    """Generate sorted triage queue"""
    
    triage_results = []
    
    for patient in patients:
        result = assign_triage_priority(patient, critical_symptoms)
        triage_results.append(result)
    
    # Sort by priority (lower number = higher priority)
    triage_results.sort(key=lambda x: x["priority"])
    
    return triage_results
